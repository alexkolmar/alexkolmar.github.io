<script>
$(document).ready(function () {
    var $announcement = $('#pun-announcement');
    if (!$announcement.length) return;

    function moveAnnouncement() {
        var $a = $('#pun-announcement');
        if (!$a.length) return;

        var isMobile = window.innerWidth <= 540;

        if (isMobile) {
            // Возвращаем внутрь .punbb перед #pun-status
            $a.insertBefore('#pun-index #pun-status');
        } else {
            // Переносим в #pun
            $a.appendTo('#pun');
        }
    }

    moveAnnouncement();
    $(window).on('resize', moveAnnouncement);
});
</script>

<script type="text/javascript">
 $(function () {
  if ($(window).scrollTop() >= "250") $("#ToTop").fadeIn("slow")
  $(window).scroll(function () {
   if ($(window).scrollTop() <= "250") $("#ToTop").fadeOut("slow")
   else $("#ToTop").fadeIn("slow")
  });
  if ($(window).scrollTop() <= $(document).height() - "999") $("#OnBottom").fadeIn("slow")
  $(window).scroll(function () {
   if ($(window).scrollTop() >= $(document).height() - "999") $("#OnBottom").fadeOut("slow")
   else $("#OnBottom").fadeIn("slow")
  });
  $("#ToTop").click(function () { $("html,body").animate({ scrollTop: 0 }, "slow") })
  $("#OnBottom").click(function () { $("html,body").animate({ scrollTop: $(document).height() }, "slow") })
 });
</script>
